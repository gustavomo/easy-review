      octokit: any,
      owner: string,
      repo: string,
      prNumber: number,
    ): Promise<PRFileChange[]> {

        repo,
        pull_number: prNumber,
        per_page: 100,  // Pitfall 5: max page size; full pagination deferred per D-06

        filename: f.filename as string,
        status: f.status as PRFileChange['status'],
        previous_filename: f.previous_filename as string | undefined,
        sha: f.sha as string,
      }));
    }

    getChildren(element?: EasyReviewTreeNode): vscode.ProviderResult<EasyReviewTreeNode[]> {
      if (!element) return this.prs;
      if (element instanceof PRTreeItem) {
        if (element.children === undefined) {
          element.children = 'loading';
          void this.loadFilesForPR(element);
          return [new LoadingNode()];

        return element.children;

      return [];
    }
