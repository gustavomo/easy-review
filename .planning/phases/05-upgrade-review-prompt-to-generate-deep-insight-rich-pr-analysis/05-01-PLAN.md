function makeOctokit(overrides: Partial<{
  listReviewComments: any;
  listReviews: any;
  listCommits: any;
}> = {}) {
  return {
    rest: {
      pulls: {
        listReviewComments: overrides.listReviewComments ?? vi.fn().mockResolvedValue({ data: [] }),
        listReviews: overrides.listReviews ?? vi.fn().mockResolvedValue({ data: [] }),

      },
    },
  };
}
